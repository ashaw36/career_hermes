## 1. 修改 src/ui/webview/api_handler.py

找到方法 `export_resume_pdf(self, persona_id: str)`（约第 637 行）。当前是占位实现，只返回 `generate_resume()` 的结果。

改为以下逻辑：

```python
    def export_resume_pdf(self, persona_id: str) -> Dict[str, Any]:
        """生成简历 PDF。先尝试调用 PDFExporter 生成真实 PDF，如果 fpdf2 未安装则降级为 Markdown 下载。"""
        try:
            import base64
            from src.services.pdf_exporter import PDFExporter, PDFExporterError
            from src.services.resume_builder import ResumeBuilder

            # 获取 Persona
            persona = AsyncRunner.run(self.persona_eng.get_by_id(persona_id))
            if not persona:
                return {"success": False, "error": "角色不存在"}

            # 通过 ResumeBuilder 获取经历列表
            builder = ResumeBuilder(persona_id=persona_id)
            AsyncRunner.run(builder.prepare())
            experiences = []
            for rew in getattr(builder, "_experiences", []) or []:
                exp = getattr(rew, "experience", None)
                if exp:
                    experiences.append(exp)

            # 生成 PDF
            exporter = PDFExporter()
            pdf_bytes = AsyncRunner.run(exporter.export_resume(persona, experiences))
            encoded = base64.b64encode(pdf_bytes).decode("utf-8")

            return {
                "success": True,
                "data": {
                    "pdf_base64": encoded,
                    "filename": f"简历_{persona.name}.pdf",
                },
            }
        except PDFExporterError:
            # fpdf2 未安装，降级为 Markdown
            md_result = self.generate_resume(persona_id)
            return {
                "success": False,
                "error": "fpdf2 未安装，PDF 导出不可用。请运行: pip install fpdf2",
                "fallback_markdown": md_result.get("markdown", "") if isinstance(md_result, dict) else "",
            }
        except Exception as e:
            logger.error(f"export_resume_pdf error: {e}")
            return {"success": False, "error": str(e)}
```

    def exportResumePDF(self, persona_id: str) -> str:
        try:
            result = self._api.export_resume_pdf(persona_id)

## 3. 修改 prototype/ui-prototype.html

找到 `exportResumePDF()` 函数（约第 1694 行）。当前下载的是 markdown 伪装的 txt 文件。

改为以下逻辑：

```javascript
    function exportResumePDF() {
      var select = document.getElementById('resume-persona-select');
      if (!select || !select.value) { showError('请先选择角色', ''); return; }
      if (!window.pybridge) return;
      window.pybridge.exportResumePDF(select.value, function(result) {
        try {
          var resp = JSON.parse(result);
          if (resp.success && resp.data && resp.data.pdf_base64) {
            // 下载真实 PDF
            var byteChars = atob(resp.data.pdf_base64);
            var byteNums = new Array(byteChars.length);
            for (var i = 0; i < byteChars.length; i++) {
              byteNums[i] = byteChars.charCodeAt(i);
            }
            var byteArray = new Uint8Array(byteNums);
            var blob = new Blob([byteArray], {type: 'application/pdf'});
            var url = URL.createObjectURL(blob);
            var a = document.createElement('a');
            a.href = url;
            a.download = resp.data.filename || ('简历_' + select.value + '.pdf');
            a.click();
            URL.revokeObjectURL(url);
          } else if (resp.fallback_markdown) {
            // 降级为 Markdown 下载
            showError('PDF导出失败', resp.error || '将降级为 Markdown 下载');
            var blob = new Blob([resp.fallback_markdown], {type: 'text/markdown'});
            var url = URL.createObjectURL(blob);
            var a = document.createElement('a');
            a.href = url;
            a.download = '简历_' + select.value + '.md';
            a.click();
            URL.revokeObjectURL(url);
          } else {
            showError('PDF导出失败', resp.error || '');
          }
        } catch (e) { showError('PDF导出错误', ''); }
      });
    }
```
