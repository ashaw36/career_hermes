```python
    async def update_reframe(
        self,
        reframe_id: str,
        reframed_summary: str,
    ) -> Optional[JobMatchExperienceReframe]:
        """
        手动更新单条重述的 reframed_summary。
        """
        async with AsyncSessionLocal() as session:
            result = await session.execute(
                select(JobMatchExperienceReframe).where(
                    JobMatchExperienceReframe.id == reframe_id
                )
            )
            reframe = result.scalar_one_or_none()
            if not reframe:
                return None
            reframe.reframed_summary = reframed_summary
            await session.commit()
            await session.refresh(reframe)
            return reframe

    async def reset_reframe(self, reframe_id: str) -> bool:
        """
        删除单条重述记录，下次会自动重新走 LLM 生成。
        """
        async with AsyncSessionLocal() as session:
            result = await session.execute(
                select(JobMatchExperienceReframe).where(
                    JobMatchExperienceReframe.id == reframe_id
                )
            )
            reframe = result.scalar_one_or_none()
            if not reframe:
                return False
            await session.delete(reframe)
            await session.commit()
            return True
```

## 2. 修改 src/ui/webview/api_handler.py

```python
    def update_reframe(self, data: Dict[str, Any]) -> Dict[str, Any]:
        """手动更新单条重述内容"""
        try:
            reframe_id = str(data.get("reframe_id", ""))
            reframed_summary = str(data.get("reframed_summary", ""))
            if not reframe_id:
                return {"success": False, "error": "缺少 reframe_id"}
            reframe = AsyncRunner.run(
                self.jd_reframe.update_reframe(reframe_id, reframed_summary)
            )
            if reframe:
                return {
                    "success": True,
                    "data": self._reframe_to_dict(reframe),
                }
            return {"success": False, "error": "重述记录不存在"}
        except Exception as e:
            logger.error(f"update_reframe error: {e}")
            return {"success": False, "error": str(e)}

    def reset_reframe(self, reframe_id: str) -> Dict[str, Any]:
        """重置单条重述，删除记录后下次重新生成"""
        try:
            if not reframe_id:
                return {"success": False, "error": "缺少 reframe_id"}
            ok = AsyncRunner.run(self.jd_reframe.reset_reframe(reframe_id))
            return {"success": ok, "data": {"deleted": ok}}
        except Exception as e:
            logger.error(f"reset_reframe error: {e}")
            return {"success": False, "error": str(e)}
```

    @Slot(str, result=str)
    def updateReframe(self, json_str: str) -> str:
        try:
            data: Dict[str, Any] = json.loads(json_str)
            result = self._api.update_reframe(data)
            return json.dumps(result, ensure_ascii=False)
        except Exception as e:
            logger.error(f"Bridge updateReframe error: {e}")
            return self._err(str(e))

    @Slot(str, result=str)
    def resetReframe(self, reframe_id: str) -> str:
        try:
            result = self._api.reset_reframe(reframe_id)
            return json.dumps(result, ensure_ascii=False)
        except Exception as e:
            logger.error(f"Bridge resetReframe error: {e}")
            return self._err(str(e))
```

    function toggleReframeEdit(reframeId) {
      var viewEl = document.getElementById('reframe-view-' + reframeId);
      var editEl = document.getElementById('reframe-edit-' + reframeId);
      if (!viewEl || !editEl) return;
      if (editEl.style.display === 'none') {
        viewEl.style.display = 'none';
        editEl.style.display = 'block';
      } else {
        viewEl.style.display = 'block';
        editEl.style.display = 'none';
      }
    }

    function saveReframeEdit(reframeId) {
      var ta = document.getElementById('reframe-ta-' + reframeId);
      if (!ta || !window.pybridge) return;
      window.pybridge.updateReframe(JSON.stringify({
        reframe_id: reframeId,
        reframed_summary: ta.value,
      }), function(result) {
        try {
          var resp = JSON.parse(result);
          if (resp.success) {
            loadReframeResults(gCurrentMatchId);
          } else {
            showError('重述保存失败', resp.error || '');
          }
        } catch (e) { showError('重述保存错误', ''); }
      });
    }

    function resetReframe(reframeId) {
      if (!confirm('确定重置这条修饰记录吗？重置后将重新走 LLM 生成。')) return;
      if (!window.pybridge) return;
      window.pybridge.resetReframe(reframeId, function(result) {
        try {
          var resp = JSON.parse(result);
          if (resp.success) {
            loadReframeResults(gCurrentMatchId);
          } else {
            showError('重述重置失败', resp.error || '');
          }
        } catch (e) { showError('重述重置错误', ''); }
      });
    }
```
